02 DATA FLOW — MESSAGE TO USER WITH REAL JSON
A concrete trace of what happens when a Telegram channel posts a new job. Every transformation is shown with the real JSON shape between stages.
1
Raw TG message arrives via grammers-client in parser worker. Pushed to Redis list queue:messages:raw.
2
Enricher worker pops the message, computes hash, checks dedup table. If new — parses budget/deadline/skills with a small LLM, generates embedding via Voyage AI, stores in messages table.
3
Matcher queries profiles where embedding cosine similarity with message > 0.65. For each candidate user, adds a row to matches (status=pending).
4
Per-user scorer picks top-K pending matches, invokes the user's own AI via AI Router with the scoring prompt. Stores final score + reason in matches, sets status=ready.
5
Notifier reads matches where status=ready, pushes to user's active WebSocket + Telegram bot + adds to morning email batch.

Email/Password Flow

use axum::{Json, extract::State};
use serde::{Deserialize, Serialize};
use argon2::{Argon2, PasswordHasher, password_hash::SaltString};

#[derive(Deserialize)]
pub struct RegisterReq {
    pub email: String,
    pub password: String,
    pub referral_code: Option<String>,
}

#[derive(Serialize)]
pub struct AuthResp {
    pub user: User,
    pub tokens: TokenPair,
}

pub async fn register(
    State(state): State<AppState>,
    Json(req): Json<RegisterReq>,
) -> Result<Json<AuthResp>, ApiError> {
    // 1. validate email format + password strength
    validate_email(&req.email)?;
    validate_password(&req.password)?;  // ≥12 chars, not in leaked list

    // 2. hash password with argon2id
    let salt = SaltString::generate(&mut rand::thread_rng());
    let hash = Argon2::default()
        .hash_password(req.password.as_bytes(), &salt)?
        .to_string();

    // 3. insert user (trial starts now, 7 days)
    let user = sqlx::query_as!(User,
        "INSERT INTO users (email, password_hash, tier, trial_ends_at)
         VALUES ($1, $2, 'trial', NOW() + INTERVAL '7 days')
         RETURNING *",
        req.email, hash
    )
    .fetch_one(&state.db)
    .await?;

    // 4. create session + issue tokens
    let tokens = state.auth.issue_tokens(user.id).await?;

    // 5. fire async welcome email, trial tracking, etc.
    state.events.send(Event::UserRegistered { user_id: user.id });

    Ok(Json(AuthResp { user, tokens }))
}

Middleware — extract user from JWT

use axum::{extract::Request, middleware::Next, response::Response};
use jsonwebtoken::{decode, Validation, DecodingKey, Algorithm};

pub async fn jwt_middleware(
    mut req: Request,
    next: Next,
) -> Result<Response, ApiError> {
    let token = req.headers()
        .get("Authorization")
        .and_then(|v| v.to_str().ok())
        .and_then(|h| h.strip_prefix("Bearer "))
        .ok_or(ApiError::Unauthorized)?;

    let data = decode::<Claims>(
        token,
        &DecodingKey::from_secret(JWT_SECRET.as_bytes()),
        &Validation::new(Algorithm::HS256),
    )?;

    // verify session still active (not revoked)
    let state: &AppState = req.extensions().get().unwrap();
    let session_valid = sqlx::query!(
        "SELECT 1 FROM sessions WHERE id = $1 AND revoked_at IS NULL",
        data.claims.sid
    )
    .fetch_optional(&state.db)
    .await?
    .is_some();

    if !session_valid {
        return Err(ApiError::Unauthorized);
    }

    req.extensions_mut().insert(data.claims);
    Ok(next.run(req).await)
}
06 BYOK ENCRYPTION FLOW ENVELOPE AES-256-GCM
Envelope encryption: a data encryption key (DEK) is generated per-user, used to encrypt the API key, then the DEK itself is encrypted with a master KMS key. This limits the blast radius — compromising the DB doesn't leak keys, compromising one DEK doesn't leak others.

Encryption (store user's API key)

use aes_gcm::{Aes256Gcm, Key, Nonce, aead::Aead};
use rand::RngCore;
use aws_sdk_kms::Client as KmsClient;

pub struct EncryptedSecret {
    pub ciphertext: Vec<u8>,          // encrypted API key
    pub nonce: [u8; 12],              // AES-GCM nonce
    pub encrypted_dek: Vec<u8>,       // DEK wrapped by KMS
    pub dek_kms_key_id: String,      // which master key to use
}

pub async fn encrypt_api_key(
    plaintext: &str,
    kms: &KmsClient,
    master_key_id: &str,
) -> Result<EncryptedSecret, CryptoError> {
    // 1. Ask KMS to generate a DEK (256-bit AES key)
    let gen = kms
        .generate_data_key()
        .key_id(master_key_id)
        .key_spec(DataKeySpec::Aes256)
        .send()
        .await?;

    let dek_plain = gen.plaintext().unwrap().as_ref();       // 32 bytes
    let dek_encrypted = gen.ciphertext_blob().unwrap().as_ref();

    // 2. Encrypt plaintext API key with DEK + random nonce
    let cipher = Aes256Gcm::new(Key::<Aes256Gcm>::from_slice(dek_plain));
    let mut nonce = [0u8; 12];
    rand::thread_rng().fill_bytes(&mut nonce);

    let ciphertext = cipher
        .encrypt(Nonce::from_slice(&nonce), plaintext.as_bytes())?;

    // 3. DEK is wiped from memory here (Drop)
    Ok(EncryptedSecret {
        ciphertext,
        nonce,
        encrypted_dek: dek_encrypted.to_vec(),
        dek_kms_key_id: master_key_id.to_string(),
    })
}

Decryption (on inference call)

pub async fn decrypt_api_key(
    secret: &EncryptedSecret,
    kms: &KmsClient,
) -> Result<SecretString, CryptoError> {
    // 1. KMS unwraps the DEK
    let dec = kms
        .decrypt()
        .ciphertext_blob(secret.encrypted_dek.clone().into())
        .key_id(&secret.dek_kms_key_id)
        .send()
        .await?;

    let dek_plain = dec.plaintext().unwrap().as_ref();

    // 2. Decrypt API key with DEK
    let cipher = Aes256Gcm::new(Key::<Aes256Gcm>::from_slice(dek_plain));
    let plaintext = cipher
        .decrypt(Nonce::from_slice(&secret.nonce), secret.ciphertext.as_ref())?;

    // 3. SecretString zeroes its memory on Drop
    Ok(SecretString::new(String::from_utf8(plaintext)?))
}
SECURITY RULES Never log plaintext keys. Never send them to Sentry/stdout/debug. Use SecretString type everywhere. Audit log every decrypt call with user_id, feature, timestamp — never the key itself. Plaintext key lives in memory only during the await of the AI provider call.

pub struct AnthropicProvider {
    client: reqwest::Client,
}

#[async_trait]
impl Provider for AnthropicProvider {
    fn name(&self) -> &str { "anthropic" }

    fn default_model(&self) -> &str { "claude-sonnet-4-6" }

    async fn chat(
        &self,
        api_key: &SecretString,
        req: ChatRequest,
    ) -> Result<ChatResponse, ProviderError> {
        let started = Instant::now();
        let model = req.model.clone().unwrap_or_else(|| self.default_model().into());

        let body = serde_json::json!({
            "model": model,
            "max_tokens": req.max_tokens,
            "temperature": req.temperature,
            "system": req.system,
            "messages": req.messages,
        });

        let resp = self.client
            .post("https://api.anthropic.com/v1/messages")
            .header("x-api-key", api_key.expose_secret())
            .header("anthropic-version", "2023-06-01")
            .json(&body)
            .timeout(Duration::from_secs(60))
            .send()
            .await?;

        if !resp.status().is_success() {
            return self.map_error(resp).await;
        }

        let data: AnthropicResp = resp.json().await?;
        let pricing = self.pricing(&model).unwrap();
        let cost = (data.usage.input_tokens as f64 / 1_000_000.0) * pricing.input_per_1m_usd
                 + (data.usage.output_tokens as f64 / 1_000_000.0) * pricing.output_per_1m_usd;

        Ok(ChatResponse {
            content: data.content.into_iter().map(|b| b.text).collect::<Vec<_>>().join("\n"),
            model_used: model,
            usage: Usage {
                input_tokens: data.usage.input_tokens,
                output_tokens: data.usage.output_tokens,
                cost_usd: cost,
            },
            latency_ms: started.elapsed().as_millis() as u64,
            finish_reason: data.stop_reason.into(),
        })
    }

    // ... embed(), validate_key(), pricing() implementations
}

Router — dispatch + budget check + logging

pub struct AiRouter {
    providers: HashMap<String, Arc<dyn Provider>>,
    db: PgPool,
    kms: KmsClient,
}

impl AiRouter {
    pub async fn chat_for_user(
        &self,
        user_id: Uuid,
        feature: &str,                 // "proposal" | "reply" | "scoring" | ...
        req: ChatRequest,
    ) -> Result<ChatResponse, RouterError> {
        // 1. Load user's credentials
        let cred = sqlx::query_as!(AiCredential,
            "SELECT * FROM ai_credentials WHERE user_id = $1 AND status = 'active' LIMIT 1",
            user_id
        )
        .fetch_optional(&self.db)
        .await?
        .ok_or(RouterError::NoCredentials)?;

        // 2. Check monthly budget cap
        let month_spend = self.get_month_spend(user_id).await?;
        if let Some(cap) = cred.monthly_budget_usd {
            if month_spend >= cap {
                return Err(RouterError::BudgetExceeded);
            }
        }

        // 3. Decrypt API key (in-memory only)
        let api_key = decrypt_api_key(&cred.to_secret(), &self.kms).await?;

        // 4. Dispatch to provider
        let provider = self.providers.get(&cred.provider).ok_or(RouterError::UnknownProvider)?;

        let resp = provider.chat(&api_key, req).await
            .map_err(RouterError::Provider)?;

        // 5. Log usage (metadata only — never content!)
        sqlx::query!(
            "INSERT INTO ai_usage (user_id, feature, provider, model, input_tokens, output_tokens, cost_usd, latency_ms) VALUES ($1, $2, $3, $4, $5, $6, $7, $8)",
            user_id, feature, cred.provider, resp.model_used,
            resp.usage.input_tokens as i32, resp.usage.output_tokens as i32,
            resp.usage.cost_usd, resp.latency_ms as i32
        )
        .execute(&self.db)
        .await?;

        Ok(resp)
    }
}

use thiserror::Error;
use axum::{response::{IntoResponse, Response}, Json, http::StatusCode};

#[derive(Debug, Error)]
pub enum ApiError {
    #[error("unauthorized")]
    Unauthorized,

    #[error("validation failed: {0}")]
    Validation(String),

    #[error("rate limited")]
    RateLimited { retry_after_s: u32 },

    #[error("AI provider not connected")]
    NoAiProvider,

    #[error("AI budget exceeded")]
    BudgetExceeded { cap_usd: f64, spent_usd: f64, resets_at: DateTime<Utc> },

    #[error("AI provider error: {0}")]
    Provider(ProviderError),

    #[error("not found")]
    NotFound,

    #[error("internal error")]
    Internal(#[from] anyhow::Error),
}

impl IntoResponse for ApiError {
    fn into_response(self) -> Response {
        let request_id = current_request_id();
        let (status, code, message, details, retryable) = match self {
            Self::Unauthorized => (
                StatusCode::UNAUTHORIZED,
                "auth.unauthorized",
                "Missing or invalid token",
                serde_json::json!({}),
                false,
            ),
            Self::RateLimited { retry_after_s } => (
                StatusCode::TOO_MANY_REQUESTS,
                "rate_limit.exceeded",
                "Too many requests",
                serde_json::json!({ "retry_after_s": retry_after_s }),
                true,
            ),
            Self::BudgetExceeded { cap_usd, spent_usd, resets_at } => (
                StatusCode::PAYMENT_REQUIRED,
                "ai_provider.budget_exceeded",
                "Monthly AI budget cap reached",
                serde_json::json!({ "cap_usd": cap_usd, "spent_usd": spent_usd, "resets_at": resets_at }),
                false,
            ),
            Self::Internal(e) => {
                // log full error for us, generic message to user
                tracing::error!(request_id = %request_id, error = ?e, "internal error");
                (
                    StatusCode::INTERNAL_SERVER_ERROR,
                    "internal",
                    "Something went wrong on our side",
                    serde_json::json!({}),
                    true,
                )
            }
            // ... other variants
        };

        (status, Json(serde_json::json!({
            "error": {
                "code": code,
                "message": message,
                "details": details,
                "request_id": request_id,
                "retryable": retryable,
            }
        }))).into_response()
    }
}
